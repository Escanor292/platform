// scripts/init-mongodb.js:
// activity_logs: userId; action; entityType+entityId; createdAt DESC; createdAt ASC TTL 90 days.
// audit_logs: entityType+entityId+createdAt DESC; userId+createdAt DESC; action+createdAt DESC;
//              sparse pgAuditLogId; createdAt ASC TTL 365 days.
// analytics_events: eventName; userId; campaignId; campaignId+eventName+createdAt DESC;
//                   createdAt DESC; createdAt ASC TTL 180 days.
// notifications: userId+createdAt DESC; userId+isRead; createdAt ASC TTL 90 days.
// campaign_comments: campaignId+createdAt DESC; campaignId+parentId+createdAt ASC;
//                    parentId+createdAt ASC; userId+createdAt DESC; status.
// campaign_updates: campaignId+status+publishedAt DESC; creatorId;
//                   campaignId+isPinned DESC+publishedAt DESC; tags.
// campaign_contents: UNIQUE campaignId+isDraft; lastSavedBy.
// user_metadata: UNIQUE userId; tags.
//
// scripts/init-chat-indexes.ts:
// conversations: UNIQUE conversationKey; participantIds; updatedAt DESC;
//                participantIds+updatedAt DESC.
// messages: conversationId+createdAt DESC; senderId; isDeleted.
// chat_reports: status+createdAt DESC; conversationId; reporterId.
//
// src/services/mongodb/blog.service.ts:
// blog_contents: UNIQUE postId.
// blog_drafts: postId; authorId.
// blog_versions: postId+versionNumber DESC.
// blog_view_logs: postId+viewedAt DESC; userId.
//
// notification.service.ts repeats notification indexes and names TTL index ttl_90_days.
// comment.service.ts repeats campaign_comments indexes. campaign-content.service.ts adds
// campaignId+isDraft at service level. user-metadata.service.ts repeats UNIQUE userId and tags.
//
// user_notes has an expiresAt field and is filtered by application logic; no code-confirmed
// TTL/index bootstrap exists for user_notes. typingBy is an in-document timestamp map and is
// not a Mongo TTL index.
//
// Feature flags: BaseMongoService-backed services (activity, audit, analytics, notifications,
// campaign updates/content, user metadata) require their ENABLE_* flag to equal the literal
// string "true". Chat and blog services access their collections directly. Index scripts are
// separate bootstrap operations and are not evidence that a live database has been initialized.
//
// No Ref declarations are emitted: MongoDB does not enforce cross-collection foreign keys;
// relationships to PostgreSQL are application-level string references documented above.
